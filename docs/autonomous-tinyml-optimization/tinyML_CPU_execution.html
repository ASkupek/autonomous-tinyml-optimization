145    def _send_to_mcu(self, connection: serial.Serial, inputs: np.ndarray) -> Optional[Tuple[float, float, float]]:
146        """Sends input features to the MCU and receives model predictions.
147
148        The input features are serialized as 32-bit floating-point values
149        and transmitted over the configured serial connection. The MCU
150        performs inference and returns three 32-bit floating-point values
151        corresponding to the configured target outputs.
152
153        Args:
154            connection (serial.Serial):
155                Active serial connection to the MCU.
156            inputs (np.ndarray):
157                Scaled input feature vector.
158
159        Returns:
160            Optional[tuple[float, float, float]]:
161                MCU predictions as (throttle, brake, steer), or None if
162                the expected response was not received.
163
164        Raises:
165            Exception:
166                If serial communication fails.
167        """
168        try:
169            header = bytes([0xAA, 0xBB])
170            payload = struct.pack("<4f",*inputs.astype(np.float32))
171            packet = header + payload
172            connection.write(packet)
173            connection.flush()
174
175            response = connection.read(4)
176
177            if len(response) != 4:
178                print(f"[Execute on MCU Warning] Expected 4 bytes, received {len(response)}.")
179                return None
180
181            return struct.unpack("<1f", response)
182
183        except Exception as error:
184            print(f"[Execute on MCU Error] MCU communication failed: {error}")
185            raise error

187    def test_on_dataset(self) -> None:
188        """Evaluates MCU model predictions against the dataset test set.
189
190        Each scaled test sample is transmitted to the MCU. The MCU performs
191        inference and returns three floating-point output values corresponding
192        to the configured target outputs.
193
194        Raises:
195            Exception: Re-raises any error encountered during dataset iteration or metric computation.
196        """  
197        try:  
198            X_test_scaled, y_test, _ = self.load_dataset()
199            window_size = self.config.pipeline.window_size
200            mcu_outputs = []
201            with serial.Serial(self.config.port_configuration.port_name, baudrate=self.config.port_configuration.baud_rate, timeout=self.config.port_configuration.timeout) as connection:
202                for i in range(0, len(y_test)):
203                    received = self._send_to_mcu(connection=connection,inputs=X_test_scaled[i])
204
205                    if received is not None:
206                        mcu_outputs.append(received)
207
208            mcu_preds_matrix = np.array(mcu_outputs)
209            mse = float(np.mean((mcu_preds_matrix - y_test) ** 2))
210            rmse = float(np.sqrt(mse))
211            mae = float(np.mean(np.abs(mcu_preds_matrix - y_test)))
212            print(f"[Execute on MCU] Evaluation Results -> MSE: {mse:.6f} | RMSE: {rmse:.6f} | MAE: {mae:.6f}")
213        except Exception as error:
214            print(f"[Execute on MCU Error] Dataset evaluation failed: {error}")
215            raise error

218    def load_dataset(self) -> Tuple[np.ndarray, pd.DataFrame, object]:
219        """Loads, splits, and scales the dataset for MCU evaluation.
220
221        The same input feature ordering defined by ``input_features`` is used
222        when constructing the model input matrix. The scaler previously fitted
223        during training is reused so that the MCU receives data using the
224        exact same scaling parameters.
225
226        Returns:
227            Tuple[np.ndarray, pd.DataFrame, object]:
228                Scaled test inputs, unscaled test targets, and fitted scaler.
229
230        Raises:
231            FileNotFoundError:
232                If the dataset or scaler cannot be found.
233            ValueError:
234                If required dataset columns are missing.
235        """
236        try:
237            if not os.path.isfile(self.config.pipeline.export_test_data):
238                raise FileNotFoundError(f"Dataset file does not exist at: {self.config.pipeline.export_test_data}")
239            # Load the dataset
240            test_data = np.load(self.config.pipeline.export_test_data)
241            X_test_scaled = test_data["X_test_scaled"]
242            y_test = test_data["y_test"]
243            scaler = joblib.load(self.config.pipeline.scaler_path)
244
245            print(f"[Execute on MCU] Raw records loaded.")
246            return (X_test_scaled, y_test, scaler)
247        except Exception as error:
248            print(f"[Execute on MCU] following error is raised during the processing of dataset {error}")
249            raise
