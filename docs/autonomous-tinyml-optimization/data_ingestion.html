106    def spawn_vehicle(self, model_name: Optional[str]='vehicle.tesla.model3') -> None:
107        """Spawns autopilot-controlled vehicles and initializes obstacle sensors.
108
109        Shuffles available map spawn points and configures the CARLA Traffic Manager
110        to prevent immediate collisions. Each vehicle gets an attached obstacle sensor.
111
112        Args:
113            model_name (str, optional): CARLA blueprint string for the car model. 
114                Defaults to 'vehicle.tesla.model3'.
115        """
116
117        num_of_cars = self.config.carla.carla_number_vehicles
118
119        carla_blueprint_library = self.carla_world.get_blueprint_library()
120        carla_spawn_points = self.carla_world.get_map().get_spawn_points()
121
122        # Shuffle points
123        random.shuffle(carla_spawn_points)
124
125        # We set this due to the reason that without the cars were spawn into the map and a lot of them crash
126        carla_traffic_manager = self.carla_client.get_trafficmanager(self.config.carla.traffic_manager_port)
127        carla_traffic_manager.set_global_distance_to_leading_vehicle(self.config.carla.traffic_manager_distance)
128
129        print(f"[CARLA] Spawning {num_of_cars} autonomous vehicles ({model_name})...")
130
131        while len(self.vehicles) < num_of_cars and len(carla_spawn_points) > 0:
132            carla_spawn_point = carla_spawn_points.pop()
133            vehicle_bp = carla_blueprint_library.find(model_name)
134
135            try:
136                vehicle = self.carla_world.spawn_actor(vehicle_bp, carla_spawn_point)
137                if not self.config.carla.mcu_testing:
138                    vehicle.set_autopilot(True, carla_traffic_manager.get_port())
139                else:
140                    vehicle.set_autopilot(False, carla_traffic_manager.get_port())
141                self.vehicles.append(vehicle)
142
143                # Set default distance
144                self.distances[vehicle.id] = self.config.carla.carla_obstacle_sensor_distance
145
146                # Adding additional sensor
147                # Obstacle sensor will look only upfront and not on back/left/right
148                obstacle_bp = carla_blueprint_library.find('sensor.other.obstacle')
149                obstacle_bp.set_attribute('distance', str(self.config.carla.carla_obstacle_sensor_reach))
150                obstacle_bp.set_attribute('hit_radius', str(self.config.carla.carla_obstacle_sensor_hit_radius))
151                obstacle_bp.set_attribute('only_dynamics', str(self.config.carla.carla_obstacle_sensor_only_dynamics))
152
153                # Attach the sensor on the vehicle
154                sensor_transform = carla.Transform(carla.Location(x=2.0, z=1.0))
155                obstacle_sensor = self.carla_world.spawn_actor(obstacle_bp, sensor_transform, attach_to=vehicle)
156
157                obstacle_sensor.listen(lambda event, v_id=vehicle.id: self._on_obstacle_detected(event=event, vehicle_id=v_id))
158                self.obstacle_detectors[vehicle.id] = obstacle_sensor
159
160                # Micro-sleep to distribute physics calculations smoothly across server frames
161                time.sleep(0.2)
162
163            except Exception as error:
164                print(f"[CARLA Warning] Failed to spawn vehicle at point: {error}")
165                continue
166        print(f"[CARLA] Successfully spawned {len(self.vehicles)}/{num_of_cars} vehicles.")

177    def get_lane_data(self, vehicle) -> tuple[float, float]:
178        """Returns the lateral offset from the lane center and the heading error.
179
180        Args:
181            vehicle (carla.Vehicle): Vehicle used to determine the current lane
182                position and heading.
183
184        Returns:
185            tuple[float, float]: A tuple containing:
186                - lane_offset (float): Lateral offset from the center of the lane.
187                - heading_error (float): Normalized heading error in range [-1.0, 1.0].
188        """
189        try:
190            transform = vehicle.get_transform()
191            location = transform.location
192
193            waypoint = self.carla_world.get_map().get_waypoint(location, project_to_road=True, lane_type=carla.LaneType.Driving)
194
195            if waypoint is None:
196                return 0.0, 0.0
197
198            vehicle_yaw = transform.rotation.yaw
199            lane_yaw = waypoint.transform.rotation.yaw
200
201            heading_error = vehicle_yaw - lane_yaw
202
203            while heading_error > 180.0:
204                heading_error -= 360.0
205
206            while heading_error < -180.0:
207                heading_error += 360.0
208
209            heading_error /= 180.0
210
211            lane_center = waypoint.transform.location
212            right_vector = waypoint.transform.get_right_vector()
213
214            dx = location.x - lane_center.x
215            dy = location.y - lane_center.y
216
217            lane_offset = (
218                dx * right_vector.x +
219                dy * right_vector.y
220            )
221
222            return lane_offset, heading_error
223        except Exception as error:
224            print(f"[Streaming Error] Unexpected error during getting lane assist: {error}")
225            return 0.0, 0.0

281    def start_streaming(self) -> None:
282        """Enters infinite telemetry extraction loop, dispatching state vectors to Kafka.
283
284        Extracts multi-axis velocity, acceleration, and manual control logs, computes 
285        scalar magnitudes, and fires JSON payloads into the pipeline telemetry stream.
286        """
287
288        sleep_time = 1.0 / self.config.pipeline.streaming_frequency_hz
289        if not self.config.carla.mcu_testing:
290            initial_time = time.time()
291        try:
292            while True:
293                if not self.config.carla.mcu_testing:
294                    if (time.time() - initial_time) >= self.config.carla.time_for_town_train:
295                        print(f"\n[Streaming] Reached target time limit of {self.config.carla.time_for_town_train / 60:.1f} minutes.")
296                        break
297
298                timestamp = time.time()
299                for vehicle in self.vehicles:
300                    if not vehicle.is_alive:
301                        continue
302
303                    vel = vehicle.get_velocity()
304                    acc = vehicle.get_acceleration()
305                    control = vehicle.get_control()
306
307                    #calculation of speed and acceleration
308                    speed = math.hypot(vel.x, vel.y, vel.z)
309                    acceleration = math.hypot(acc.x, acc.y, acc.z)
310                    lane_offset, heading_error = self.get_lane_data(vehicle=vehicle)
311                    message = {
312                        "timestamp": timestamp,
313                        "vehicle_id": vehicle.id,
314                        "speed": speed,
315                        "acceleration": acceleration,
316                        "throttle": control.throttle,
317                        "brake": control.brake,
318                        "steer": control.steer,
319                        "distance": self.distances.get(vehicle.id, self.config.carla.carla_obstacle_sensor_distance),
320                        "friction": 1.0,
321                        "lane_offset": lane_offset,
322                        "heading_error": heading_error                
323                    }
324
325                    if self.kafka_producer is not None:
326                        payload = json.dumps(message).encode('utf-8')
327                        self.kafka_producer.produce(
328                            self.config.kafka.kafka_topic, 
329                            value=payload,
330                            on_delivery = self.on_delivery)
331                # Clear network buffer
332                if self.kafka_producer is not None:
333                    self.kafka_producer.poll(0)
334
335                time.sleep(sleep_time)
336
337        except KeyboardInterrupt:
338            print("\n[Streaming] User interruption detected (Ctrl+C). Stopping loop...")
339        except Exception as error:
340            print(f"[Streaming Error] Unexpected error during streaming: {error}")
341        #finally:
342            #self.cleanup()

344    def cleanup(self) -> None:
345        """Gracefully purges execution environment, destroying CARLA actors and flushing Kafka."""
346        print("[Cleanup] Terminating sensors and vehicle actors...")
347
348        for sensor in self.obstacle_detectors.values():
349            if sensor is not None and sensor.is_alive:
350                try:
351                    sensor.stop()
352                    sensor.destroy()
353                except Exception as error:
354                    print(f"[Cleanup Warning] Error destroying sensor: {error}")
355
356        for vehicle in self.vehicles:
357            if vehicle is not None and vehicle.is_alive:
358                try:
359                    vehicle.destroy()
360                except Exception as error:
361                    print(f"[Cleanup Warning] Error destroying vehicle {vehicle.id}: {error}")
362                    pass
363        self.vehicles.clear()
364        self.obstacle_detectors.clear()
365        self.distances.clear()
366        if not self.config.carla.mcu_testing:
367            if self.kafka_producer is not None:
368                print("[Kafka] Flushing remaining telemetry messages...")
369                self.kafka_producer.flush()
370                
371                #Old versions didnt have close
372                if hasattr(self.kafka_producer, "close"):
373                    self.kafka_producer.close()
374        print("[Cleanup] Cleanup complete.")

def get_lane_data(self, vehicle) -> tuple[float, float]: 
177    def get_lane_data(self, vehicle) -> tuple[float, float]:
178        """Returns the lateral offset from the lane center and the heading error.
179
180        Args:
181            vehicle (carla.Vehicle): Vehicle used to determine the current lane
182                position and heading.
183
184        Returns:
185            tuple[float, float]: A tuple containing:
186                - lane_offset (float): Lateral offset from the center of the lane.
187                - heading_error (float): Normalized heading error in range [-1.0, 1.0].
188        """
189        try:
190            transform = vehicle.get_transform()
191            location = transform.location
192
193            waypoint = self.carla_world.get_map().get_waypoint(location, project_to_road=True, lane_type=carla.LaneType.Driving)
194
195            if waypoint is None:
196                return 0.0, 0.0
197
198            vehicle_yaw = transform.rotation.yaw
199            lane_yaw = waypoint.transform.rotation.yaw
200
201            heading_error = vehicle_yaw - lane_yaw
202
203            while heading_error > 180.0:
204                heading_error -= 360.0
205
206            while heading_error < -180.0:
207                heading_error += 360.0
208
209            heading_error /= 180.0
210
211            lane_center = waypoint.transform.location
212            right_vector = waypoint.transform.get_right_vector()
213
214            dx = location.x - lane_center.x
215            dy = location.y - lane_center.y
216
217            lane_offset = (
218                dx * right_vector.x +
219                dy * right_vector.y
220            )
221
222            return lane_offset, heading_error
223        except Exception as error:
224            print(f"[Streaming Error] Unexpected error during getting lane assist: {error}")
225            return 0.0, 0.0

 76    def kafka_start_listening(self) -> None:
 77        """ Enters an infinite polling loop to extract, decode, and buffer incoming telemetry.
 78            Continuously polls the Kafka topic for new messages. When the internal memory
 79            buffer reaches `kafka_buffer_size`, it automatically triggers `flush_buffer()`.
 80            Handles KeyboardInterrupt gracefully to clean up connections.
 81        """
 82
 83        if not self.kafka_consumer:
 84            self.kafka_connection()
 85        
 86        self.is_running = True
 87
 88        try:
 89            while self.is_running:
 90
 91                msg = self.kafka_consumer.poll(timeout=1.0)
 92
 93                if msg is None:
 94                    continue
 95                
 96                if msg.error():
 97                    if msg.error().code() == KafkaError._TIMED_OUT:
 98                        continue
 99                    else:
100                        print(f"[Kafka Consumer Error] Stream error: {msg.error()}")
101                        continue
102
103                try:
104                    raw_payload = msg.value()
105                    if not raw_payload:
106                        continue
107                    data = json.loads(raw_payload.decode('utf-8'))
108                except Exception as parse_error:
109                    print(f"[Kafka Consumer Error] Payload parsing failed: {parse_error}")
110                    continue
111
112                filtered_row = {col: data[col] for col in self.received_data if col in data}
113
114                if len(filtered_row) == len(self.received_data):
115                    self.kafka_buffer.append(filtered_row)
116                    
117                if len(self.kafka_buffer) >= self.kafka_buffer_size:
118                    self.flush_buffer()
119        except KeyboardInterrupt:
120            print("\n[Kafka Consumer] User interruption detected (Ctrl+C). Stopping loop...")
121        except Exception as error:
122            print(f"[Kafka Consumer Error] Unexpected error during event loop: {error}")
123        finally:
124            self.stop_listening()

145        except Exception as error:
