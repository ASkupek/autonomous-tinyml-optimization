 71    def _compute_weighted_loss(self, outputs: torch.Tensor, y_true: torch.Tensor) -> torch.Tensor:
 72        """Calculates a custom weighted Mean Squared Error (MSE) loss.
 73
 74        Samples with larger ground-truth steering angles receive higher weights.
 75        This prioritizes accurate predictions during turns and recovery maneuvers
 76        over straight-line driving.
 77
 78
 79        Args:
 80            outputs (torch.Tensor): Predicted model outputs of shape (batch_size, num_outputs).
 81            y_true (torch.Tensor): Ground-truth target values of shape (batch_size, num_outputs).
 82
 83        Returns:
 84            torch.Tensor: Scalar tensor representing the mean weighted loss for the batch.
 85        """
 86        y_true_reshaped = y_true.view_as(outputs)
 87
 88        # Squared error between predicted and true values
 89        error = torch.square(outputs - y_true_reshaped)
 90
 91        # absolute value of the true steering angle to determine the weight for each sample
 92        abs_y = torch.abs(y_true_reshaped)
 93
 94        
 95        # Steering samples are weighted according to their magnitude.
 96        # The weight increases linearly from 1.0 at |steer| = 0.0
 97        # to 3.0 at |steer| = 0.1 and is capped at 3.0.
 98        #
 99        # |steer| = 0.00 -> weight = 1.0
100        # |steer| = 0.05 -> weight = 2.0
101        # |steer| >= 0.10 -> weight = 3.0
102        weights = 1.0 + 2.0 * torch.clamp(
103            abs_y / 0.1,
104            min=0.0,
105            max=1.0
106        )
107        # Return weighted MSE
108        return torch.mean(weights * error)

185        # This is useful because we want to make sure that the model is well calibrated for all steering angles, especially for sharp turns and recovery maneuvers.
186

189            all_inputs = []
190            all_targets = []
191
192            for inputs, targets in validation_loader:
193                all_inputs.append(inputs.cpu())
194                all_targets.append(targets.cpu())
195

197            all_targets = torch.cat(all_targets, dim=0)

200            abs_targets = torch.abs(all_targets)
201
202            # Define steering ranges
203            masks = [
204                abs_targets <= 0.01,
205                (abs_targets > 0.01) & (abs_targets <= 0.05),

212                5000,
213                2000,
214                2000,
215                1000,
